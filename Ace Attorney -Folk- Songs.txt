    // Function to process and make icons clickable
    function processIcons() {
        var icons = document.querySelectorAll('.character');

        if (icons.length > 0) {
            console.log("Icons found:", icons.length);

            icons.forEach(function(icon) {
                var iconId = icon.getAttribute('id'); // Get the id of the icon

                // Check if the icon's ID exists in the URL map
                if (urlMap[iconId]) {
                    console.log("Assigning URL to icon with ID:", iconId);

                    // Make the icon clickable, open the corresponding URL
                    icon.addEventListener('click', function() {
                        window.open(urlMap[iconId], '_blank'); // Open the URL in a new tab
                    });
                }
            });
        }
    }

    // Observe for dynamically added elements
    var observer = new MutationObserver(function(mutations) {
        mutations.forEach(function(mutation) {
            if (mutation.addedNodes.length > 0) {
                processIcons(); // Process icons when new elements are added
            }
        });
    });

    // Start observing the body for changes
    observer.observe(document.body, { childList: true, subtree: true });

    // Run the function initially in case the icons are already present
    processIcons();
})();
